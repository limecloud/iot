cmake_minimum_required(VERSION 3.22)
set(EXTRA_COMPONENT_DIRS "../../components")
set(COMPONENTS main)
set(LIMEIOT_FIRMWARE_VERSION "0.3.0" CACHE STRING "LimeIoT firmware version")
if(NOT LIMEIOT_FIRMWARE_VERSION MATCHES "^(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)\\.(0|[1-9][0-9]*)$")
    message(FATAL_ERROR "LIMEIOT_FIRMWARE_VERSION must be major.minor.patch")
endif()
string(LENGTH "${LIMEIOT_FIRMWARE_VERSION}" version_length)
if(version_length GREATER 31)
    message(FATAL_ERROR "LIMEIOT_FIRMWARE_VERSION exceeds the native image field")
endif()
set(PROJECT_VER "${LIMEIOT_FIRMWARE_VERSION}")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(limeiot_desk_node VERSION ${LIMEIOT_FIRMWARE_VERSION})
